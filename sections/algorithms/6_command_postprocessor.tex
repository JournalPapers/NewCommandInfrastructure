\subsection{CommandPostProcessor} 
\begin{algorithm}[H]
\caption{CommandPostProcessor (\emph{Interface})}
\begin{algorithmic}[1]
\Procedure{Run}{command : Command\textlangle T\textrangle} 
    \State \textbf{Input:} command of type \textlangle T\textrangle
    \State \textbf{Output:} none

\EndProcedure
\end{algorithmic}
\end{algorithm}
